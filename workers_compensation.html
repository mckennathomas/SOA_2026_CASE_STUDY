Comparing Severity Distributions: Lognormal vs Gamma

# Lognormal AIC
loglik_logn = sum(dlnorm(x, meanlog = mu_raw, sdlog = sigma_raw, log = TRUE))
aic_logn    = 2*2 - 2*loglik_logn

# Gamma MOM estimates + AIC
m         = mean(x); v = var(x)
shape_hat = m^2 / v
rate_hat  = m / v

loglik_gam = sum(dgamma(x, shape = shape_hat, rate = rate_hat, log = TRUE))
aic_gam    = 2*2 - 2*loglik_gam

delta_aic = aic_gam - aic_logn

Part 4: Monte Carlo Simulation by Territory

Simulation Function

simulate_system = function(E_period, f_mult, s_mult, mu_base, sigma, 
                            n_sims = 30000, seed = 42) {
  set.seed(seed)
  
  if (any(is.na(c(f_mult, s_mult)))) {
    return(c(Expected_Loss = NA, Std_Dev = NA, Variance = NA, VaR_99 = NA, TVaR_99 = NA))
  }
  
  # Adjusted expected claim count for this territory
  mu_period_adj = as.numeric(lambda_hat_nb) * f_mult * E_period
  
  agg = numeric(n_sims)
  
  for (i in seq_len(n_sims)) {
    # Frequency: NegBin with fitted theta and territory-adjusted mu
    N = rnbinom(1, size = theta_nb, mu = mu_period_adj)
    
    if (N == 0) {
      agg[i] = 0
    } else {
      # Severity: LogNormal with regression-derived mu and sigma
      # s_mult shifts the log-mean multiplicatively (log scale addition)
      vals = rlnorm(N, meanlog = mu_base + log(s_mult), sdlog = sigma)
      vals[is.infinite(vals)] = 1e8   # cap extreme outliers for stability
      agg[i] = sum(vals)
    }
  }
  
  # Risk metrics
  sim_mean = mean(agg, na.rm = TRUE)
  sim_var  = var(agg, na.rm = TRUE)
  sim_sd   = sd(agg, na.rm = TRUE)
  VaR_99   = as.numeric(quantile(agg, 0.99, na.rm = TRUE, names = FALSE))
  TVaR_99  = if (is.na(VaR_99)) NA else mean(agg[agg >= VaR_99], na.rm = TRUE)
  
  c(Expected_Loss = sim_mean, 
    Std_Dev       = sim_sd,
    Variance      = sim_var,
    VaR_99        = VaR_99, 
    TVaR_99       = TVaR_99)
}

# Join and Price
work_comp_pricing_final <- work_comp_freq %>%
  left_join(pol_sev_features, by = "policy_id") %>%
  mutate(
    # Fill NAs using the global medians we calculated in Part 1/2
    avg_gear = replace_na(avg_gear, median(work_comp_sev_clean$protective_gear_quality, na.rm = TRUE)),
    avg_stress = replace_na(avg_stress, median(work_comp_sev_clean$psych_stress_index, na.rm = TRUE)),
    occ_clean = replace_na(occ_clean, "Other"),
    solar_system_clean = replace_na(solar_system_clean, "Unknown")
  ) %>%
  mutate(
    # ---  GEAR CREDIT (Proven by -0.1488 Cor) ---
    m1 = ifelse(avg_gear >= 4.0, 0.88, 1.00),
    
    # ---  STRESS SURCHARGE (Proven by 0.2441 Cor) ---
    m2 = ifelse(avg_stress >= 7.5, 1.15, 1.00),
    
    # ---  OCC SUBLIMIT (Proven by Executive/Manager Tail Risk) ---
    m3 = ifelse(occ_clean %in% c("Executive", "Manager"), 0.95, 1.00),
    
    # ---  TERRITORY (Proven by Helionis Cluster TVaR) ---
    m4 = ifelse(solar_system_clean == "Helionis", 1.23, 1.00),
    
    # --- EXPERIENCE (Proven by Mid-Level U-Shape) ---
    m5 = case_when(
      experience_yrs > 2 & experience_yrs <= 10 ~ 0.90,  # Mid-level discount
      experience_yrs <= 2 ~ 1.10,                        # Rookie surcharge
      TRUE ~ 1.05                                        # Veteran surcharge
    ),
    
    # Base Rate: $150 (Derived fromlambda_hat and expected loss)
    Base_Rate = 150.00,
    
    # FINAL PREMIUM CALCULATION
    Final_Premium = round(exposure * Base_Rate * m1 * m2 * m3 * m4 * m5, 2)
  )

# 3. Summary Table to verify it matches your Monte Carlo "out" values
pricing_summary <- work_comp_pricing_final %>%
  group_by(solar_system_clean) %>%
  summarise(
    Policies = n(),
    Total_Exposure = sum(exposure, na.rm = TRUE),
    Total_Premium = sum(Final_Premium, na.rm = TRUE),
    Avg_Premium = mean(Final_Premium / exposure, na.rm = TRUE)
  )

print(pricing_summary)
