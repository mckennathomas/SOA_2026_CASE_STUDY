agg_losses <- numeric(n_sims)

for (i in seq_len(n_sims)) {
  N <- rnbinom(1, size = theta_nb, mu = mu_period)

    agg_losses[i] <- 0

simulate_system <- function(E_period, f_mult, s_mult, n_sims = 30000, seed = 123) {
  set.seed(seed)
  
  # Ensure we are working with clean numbers, not named vectors
  f_mult <- as.numeric(f_mult)
  s_mult <- as.numeric(s_mult)
  
  # Calculate the expected number of claims for the period
  # lambda_hat_nb should be your baseline frequency from the NB model
  mu_period <- as.numeric(lambda_hat_nb) * f_mult * E_period

  # Step 1: Generate N (number of claims) for all simulations at once
  # This is much faster than a for-loop
  N_sims <- rnbinom(n_sims, size = theta_nb, mu = mu_period)
  
  # Step 2: Generate aggregate losses
  agg <- sapply(N_sims, function(n) {
    if (n == 0) return(0)
    # Applying the severity multiplier to the meanlog
    sum(rlnorm(n, meanlog = mu + log(s_mult), sdlog = sigma))
  })

  # Step 3: Calculate Metrics
  VaR_99  <- quantile(agg, 0.99, names = FALSE)
  TVaR_99 <- mean(agg[agg >= VaR_99])

  c(E_L = mean(agg), Var_L = var(agg), VaR_99 = VaR_99, TVaR_99 = TVaR_99)
}
